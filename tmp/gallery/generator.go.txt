// TEMPORARY preview generator. Not part of the suite: it writes a static HTML
// gallery of the real render SVGs over the flat green fixture so the design can
// be reviewed without Docker.
func loadFace(t *testing.T, rel string) *sfnt.Font {
	t.Helper()
	path, err := filepath.Abs(rel)
	if err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	f, err := sfnt.Parse(data)
	if err != nil {
		t.Fatal(err)
	}
	return f
}

var attrRE = regexp.MustCompile(`([a-z-]+)="([^"]*)"`)

// inkBox walks the string with the bundled face's own advances and glyph boxes,
// which is what resvg reports back from --query-all.
func inkBox(f *sfnt.Font, text string, tracking float64) (x, top, w, h float64) {
	var b sfnt.Buffer
	ppem := fixed.I(100)
	pen := 0.0
	minX, minY, maxX, maxY := 1e9, 1e9, -1e9, -1e9
	for _, r := range text {
		idx, err := f.GlyphIndex(&b, r)
		if err != nil || idx == 0 {
			pen += 50 + tracking
			continue
		}
		bounds, advance, err := f.GlyphBounds(&b, idx, ppem, font.HintingNone)
		if err != nil {
			pen += 50 + tracking
			continue
		}
		x0, y0 := pen+float64(bounds.Min.X)/64, float64(bounds.Min.Y)/64
		x1, y1 := pen+float64(bounds.Max.X)/64, float64(bounds.Max.Y)/64
		if x1 > x0 {
			minX, maxX = min(minX, x0), max(maxX, x1)
			minY, maxY = min(minY, y0), max(maxY, y1)
		}
		pen += float64(advance)/64 + tracking
	}
	if maxX <= minX {
		return 0, -70, 1, 1
	}
	return minX, minY, maxX - minX, maxY - minY
}

func galleryRenderer(t *testing.T) (*Adapter, *Rendering) {
	t.Helper()
	pretendard := loadFace(t, "../../../assets/fonts/pretendard/PretendardVariable.ttf")
	paperlogy := loadFace(t, "../../../assets/fonts/paperlogy/Paperlogy-8ExtraBold.ttf")
	a := newAdapter(t, &fakeRunner{run: func(_ context.Context, c Command) ([]byte, error) {
		data, err := os.ReadFile(c.Args[len(c.Args)-1])
		if err != nil {
			return nil, err
		}
		out := ""
		for i, part := range strings.Split(string(data), `id="m`)[1:] {
			head := part[:strings.Index(part, ">")]
			raw := part[strings.Index(part, ">")+1 : strings.Index(part, "</text>")]
			text := html.UnescapeString(raw)
			family, tracking := "", 0.0
			for _, m := range attrRE.FindAllStringSubmatch(head, -1) {
				switch m[1] {
				case "font-family":
					family = m[2]
				case "letter-spacing":
					tracking, _ = strconv.ParseFloat(m[2], 64)
				}
			}
			face := pretendard
			if family == design.FontFamily("paperlogy") {
				face = paperlogy
			}
			x, top, w, h := inkBox(face, text, tracking)
			out += fmt.Sprintf("m%d,%.3f,%.3f,%.3f,%.3f\n", i, x, 200+top, w, h)
		}
		return []byte(out), nil
	}})
	return a, testRenderer(t, a)
}

	page := &strings.Builder{}

		plan := declaredPlan(t, s.body, s.ratio)
		plan.Accent = "coral"
		var layout declaredLayout
		if err := a.WithWorkspace(t.Context(), "gallery", func(ws clip.MediaWorkspace) error {
			var err error
			layout, err = r.layoutComposition(t.Context(), ws, plan)

			frame := &strings.Builder{}
			fmt.Fprintf(frame, `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 %d %d" width="%d" height="%d"><rect width="%d" height="%d" fill="#1F7A3D"/>`, canvas.Width, canvas.Height, canvas.Width, canvas.Height, canvas.Width, canvas.Height)

				shown = append(shown, visual.manifest.Role)
			}
			frame.WriteString("</svg>")
			for _, visual := range layout.visuals {
				if visual.manifest.StartMS > at || visual.manifest.EndMS <= at {
					continue
				}
				parts := []design.Element{}
				for _, p := range visual.manifest.Parts {
					if p.Kind == "copy" || p.Kind == "plate" {
						parts = append(parts, p)
					}
				}
				for i := range parts {
					for j := i + 1; j < len(parts); j++ {
						a, b := parts[i].Region, parts[j].Region

				}
			}

			if err := os.WriteFile(filepath.Join(out, name), []byte(frame.String()), 0o644); err != nil {

			}
