import { create } from '@bufbuild/protobuf'
import { createRouterTransport } from '@connectrpc/connect'
import { act, cleanup, renderHook, waitFor } from '@testing-library/react'
import { afterEach, expect, it } from 'vitest'
import { SpokenVoiceGenerationService as Service } from '@/shared/api'
import { createTestQueryClient, withProviders } from '@/test/session'
import { useSpokenOperation } from '@/entities/spoken-voice/api/work'
import { SPOKEN_POLL_MS } from '@/entities/spoken-voice/config/limits'

afterEach(cleanup)
it('stops polling at received and misses normal subsequent confirmation publication', async () => {
  let state = 'received'
  let reads = 0
  const transport = createRouterTransport(({ rpc }) => {
    rpc(Service.method.getSpokenOperation, () => {
      reads++
      return create(Service.method.getSpokenOperation.output, {
        operation: { id: 'operation', kind: 'voice_confirm', state, jobId: 'job', draftId: 'draft', resultId: state === 'published' ? 'voice' : '' },
      })
    })
  })
  const cache = createTestQueryClient()
  const view = renderHook(() => useSpokenOperation('alice', 'operation'), {
    wrapper: withProviders(transport, cache),
  })
  await waitFor(() => expect(view.result.current.operation?.state).toBe('received'))
  state = 'published' // normal backend SaveConfirmation transition after its durable received journal
  await act(async () => { await new Promise(resolve => setTimeout(resolve, SPOKEN_POLL_MS * 2 + 100)) })
  expect(reads).toBe(1)
  expect(view.result.current.operation?.state).toBe('received')
  await act(async () => { await view.result.current.refetch() })
  expect(reads).toBe(2)
  expect(view.result.current.operation?.state).toBe('published')
})
