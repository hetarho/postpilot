function Editor({ slug }: { slug: string }) {
  const { post } = usePost(slug)
  return post?.content ? <BlockEditor key={`${post.slug}:${post.machineBaselineRevision}`} post={post} /> : null
}
function mount(content: PostContent, images: FakePostRow['images'] = []) {
  const contentSaves: Array<{slug: string; expectedRevision: bigint; content: PostContent}> = []
  const transport = createFakePostsTransport({ posts: [{slug:'audit-post', status:'review', content, images, contentRevision:1n, machineBaselineRevision:1n}], contentSaves })
  render(<Editor slug="audit-post" />, {wrapper:withProviders(transport,createTestQueryClient())})
  return contentSaves
}

it('reproduces typing during URL save admitting a second write and an older delayed write overriding it', async () => {
  const user = userEvent.setup()
  let releaseOld!:()=>void
  const oldGate = new Promise<void>(resolve=>{releaseOld=resolve})
  const saves: string[] = []
  const post = create(PostSchema,{slug:'url-post',status:'finalized',targetLanguage:contentLanguageToProto('ko')})
  const transport = createRouterTransport(({rpc})=>{
    rpc(PostService.method.getPost,()=>create(GetPostResponseSchema,{post}))
    rpc(PostService.method.savePostPublishedUrl,async req=>{
      saves.push(req.url)
      if(saves.length===1) await oldGate
      post.publishedUrl=req.url
      post.status='published'
      return create(SavePostPublishedUrlResponseSchema,{post})
    })
  })
  function Field(){const {post}=usePost('url-post'); return post ? <PublishedUrlField post={post}/> : null}
  render(<Field/>,{wrapper:withProviders(transport,createTestQueryClient())})
  const field = await screen.findByLabelText('네이버 블로그 글 주소')
  await user.type(field,'https://blog.naver.com/alice/1')
  await user.click(screen.getByRole('button',{name:'저장'}))
  expect(screen.getByRole('button',{name:'저장'})).toBeDisabled()
  await user.clear(field)
  await user.type(field,'https://blog.naver.com/alice/2')
  expect(screen.getByRole('button',{name:'저장'})).toBeEnabled()
  await user.click(screen.getByRole('button',{name:'저장'}))
  await waitFor(()=>expect(saves).toHaveLength(2))
  expect(await screen.findByLabelText('네이버 블로그 글 주소')).toHaveValue('https://blog.naver.com/alice/2')
  releaseOld()
  await waitFor(()=>expect(screen.getByLabelText('네이버 블로그 글 주소')).toHaveValue('https://blog.naver.com/alice/1'))
})

it('reproduces applying a different catalog document than its delayed first preview displayed', async () => {
  const user = userEvent.setup()
  let release!:()=>void
  const gate = new Promise<void>(resolve=>{release=resolve})
  const previews: string[] = [], applies: string[] = []
  const documentA='# postpilot models v1\n[writing]\nprovider/model-a premium\n'
  const documentB='# postpilot models v1\n[writing]\nprovider/model-b value\n'
  const transport = createRouterTransport(({rpc})=>{
    rpc(ModelCatalogService.method.exportCatalogDocument,()=>create(ExportCatalogDocumentResponseSchema,{document:'# postpilot models v1\n[writing]\nprovider/original'}))
    rpc(ModelCatalogService.method.previewCatalogDocument,async req=>{previews.push(req.document);await gate;return create(PreviewCatalogDocumentResponseSchema,{purposes:[{purpose:'writing',register:['provider/model-a'],deregister:['provider/original']}]})})
    rpc(ModelCatalogService.method.applyCatalogDocument,req=>{applies.push(req.document);return create(ApplyCatalogDocumentResponseSchema,{applied:true,purposes:[{purpose:'writing',register:['provider/model-b'],deregister:['provider/original']}]})})
  })
  render(<CatalogDocumentPanel open onClose={()=>{}}/>,{wrapper:withProviders(transport,createTestQueryClient())})
  fireEvent.change(screen.getByLabelText('붙여넣기'),{target:{value:documentA}})
  await user.click(screen.getByRole('button',{name:'미리보기'}))
  await waitFor(()=>expect(previews).toEqual([documentA]))
  fireEvent.change(screen.getByLabelText('붙여넣기'),{target:{value:documentB}})
  release()
  await screen.findByText('적용하면 이렇게 바뀌어요')
  expect(screen.getByText('provider/model-a')).toBeInTheDocument()
  expect(screen.queryByText('provider/model-b')).not.toBeInTheDocument()
  expect(screen.getByRole('button',{name:'확정'})).toBeEnabled()
  await user.click(screen.getByRole('button',{name:'확정'}))
  await waitFor(()=>expect(applies).toEqual([documentB]))
  expect(previews).toEqual([documentA])
})
