 const splitTypes={name:'review-control-split-type-imports',enforce:'pre',apply:'build',transform(code,id){
  if(!id.endsWith('/app/routes/models.ts'))return null;

  });
 }};
 return {...config,plugins:[splitTypes,...config.plugins],server:{...config.server,fs:{allow:['/private/tmp/postpilot-review-all-261008','/Users/ihaelam/project/postpilot']}}};
}
