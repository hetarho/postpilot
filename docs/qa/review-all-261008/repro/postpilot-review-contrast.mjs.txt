import {chromium} from '/Users/ihaelam/project/postpilot/node_modules/playwright/index.mjs';import {writeFileSync} from 'node:fs';
const browser=await chromium.launch({headless:true});const page=await browser.newPage();await page.goto('http://localhost:28961');const result=await page.evaluate(()=>{const result=[];const c=document.createElement('canvas');c.width=c.height=1;const ctx=c.getContext('2d');function rgb(name){ctx.clearRect(0,0,1,1);const color=getComputedStyle(document.documentElement).getPropertyValue(name).trim();if(!color)return null;ctx.fillStyle=color;ctx.fillRect(0,0,1,1);return [...ctx.getImageData(0,0,1,1).data]}function luminance(a){return a.slice(0,3).map(v=>v/255).map(v=>v<=.04045?v/12.92:((v+.055)/1.055)**2.4).reduce((sum,v,i)=>sum+v*[.2126,.7152,.0722][i],0)}const pairs=[];for(const fg of ['content-primary','content-secondary','content-tertiary'])for(const bg of ['surface-base','surface-recessed','surface-raised','surface-highest','surface-lowest','surface-overlay'])pairs.push([fg,bg]);for(const group of ['button-cta','button-secondary','button-ghost','button-danger-quiet','field','notice-danger','notice-success','notice-warning','notice-info','badge-accent'])pairs.push([group+'-fg',group+'-bg']);for(const theme of ['day','night']){document.documentElement.dataset.theme=theme;for(const [fg,bg] of pairs){const f=rgb('--'+fg),b=rgb('--'+bg);if(!f||!b||!f[3]||!b[3])continue;const lf=luminance(f),lb=luminance(b);result.push({theme,fg,bg,ratio:Math.round(((Math.max(lf,lb)+.05)/(Math.min(lf,lb)+.05))*100)/100,rgba:[f,b]});}}return result});writeFileSync('/private/tmp/postpilot-review-contrast-proof.json',JSON.stringify(result,null,2));console.log(JSON.stringify({pairs:result.length,below4_5:result.filter(x=>x.ratio<4.5)},null,2));await browser.close();
